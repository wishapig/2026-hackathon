cmake_minimum_required(VERSION 3.16)
project(compiler2026_dag LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
option(DAG_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(DAG_TSAN "Enable thread sanitizer (separate build)" OFF)
option(DAG_REQUIRE_OFFICIAL "Refuse local debug kernels for submission builds" OFF)
set(DAG_OFFICIAL_LIBRARY "" CACHE FILEPATH "Official judge operator archive; empty builds demo kernels for local debugging only")
if(DAG_REQUIRE_OFFICIAL AND NOT DAG_OFFICIAL_LIBRARY)
  message(FATAL_ERROR "Submission build requires DAG_OFFICIAL_LIBRARY")
endif()
if(DAG_SANITIZE AND DAG_TSAN)
  message(FATAL_ERROR "Choose ASAN/UBSAN or TSAN, not both")
endif()
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(ZLIB 1.2.9 REQUIRED)
set(DAG_METADATA_ENCODING "auto" CACHE STRING "Metadata payload encoding: auto, raw, zlib, compact, compact-zlib, binary, binary-zlib, lean, or lean-deflate")
set_property(CACHE DAG_METADATA_ENCODING PROPERTY STRINGS auto raw zlib compact compact-zlib binary binary-zlib lean lean-deflate)
if(NOT DAG_METADATA_ENCODING MATCHES "^(auto|raw|zlib|compact|compact-zlib|binary|binary-zlib|lean|lean-deflate)$")
  message(FATAL_ERROR "DAG_METADATA_ENCODING must be auto, raw, zlib, compact, compact-zlib, binary, binary-zlib, lean, or lean-deflate")
endif()
add_compile_options(-Wall -Wextra -Wconversion -Wshadow -fno-lto -fno-fast-math)
add_link_options(-fno-lto)
if(DAG_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(DAG_TSAN)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
add_library(dag_runtime STATIC src/metadata.cpp src/graph.cpp src/runtime.cpp)
add_library(dag_estimator_core STATIC src/metadata.cpp src/graph.cpp)
target_include_directories(dag_runtime PUBLIC include)
target_include_directories(dag_estimator_core PUBLIC include)
target_link_libraries(dag_runtime PUBLIC Threads::Threads ZLIB::ZLIB)
target_link_libraries(dag_estimator_core PUBLIC ZLIB::ZLIB)
if(DAG_OFFICIAL_LIBRARY)
  if(NOT EXISTS "${DAG_OFFICIAL_LIBRARY}")
    message(FATAL_ERROR "Official library does not exist")
  endif()
  add_library(contest_kernels STATIC IMPORTED GLOBAL)
  set_target_properties(contest_kernels PROPERTIES IMPORTED_LOCATION "${DAG_OFFICIAL_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/vendor/demo/include")
  set(DAG_KERNEL_MODE "official-external")
else()
  add_library(contest_kernels STATIC vendor/demo/src/base_kernels/kernels_public.cpp vendor/demo/src/base_kernels/kernels_impl.cpp)
  target_include_directories(contest_kernels PUBLIC vendor/demo/include PRIVATE vendor/demo/src/base_kernels)
  set(DAG_KERNEL_MODE "demo-local-debug")
  message(STATUS "LOCAL DEBUG: building provided demo kernels. Official submission must set DAG_OFFICIAL_LIBRARY.")
endif()
add_executable(dag_run src/main.cpp)
target_link_libraries(dag_run PRIVATE dag_runtime contest_kernels)
target_compile_definitions(dag_run PRIVATE DAG_KERNEL_MODE="${DAG_KERNEL_MODE}")
target_link_options(dag_run PRIVATE "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/dag_run.map")
add_executable(baseline_serial vendor/demo/src/baseline/main.cpp vendor/demo/src/baseline/block_cholesky.cpp)
target_link_libraries(baseline_serial PRIVATE contest_kernels)
add_executable(spd_generator vendor/public20/tools/local_generator/spd_generator.cpp)
target_include_directories(spd_generator PRIVATE vendor/demo/include)
target_link_libraries(spd_generator PRIVATE Threads::Threads)
add_executable(public_verifier vendor/public20/tools/local_verifier/verifier.cpp)
target_include_directories(public_verifier PRIVATE vendor/demo/include)
target_link_libraries(public_verifier PRIVATE Threads::Threads)
add_executable(dag_estimate tools/dag_estimate.cpp)
target_link_libraries(dag_estimate PRIVATE dag_estimator_core contest_kernels)
target_include_directories(dag_estimate PRIVATE vendor/demo/include)
add_executable(dag_check tools/check_result.cpp)
target_include_directories(dag_check PRIVATE vendor/demo/include include)
target_link_libraries(dag_check PRIVATE Threads::Threads)
find_program(DAG_LLVM_COMPILER NAMES clang++ clang)
if(NOT DAG_LLVM_COMPILER)
  message(FATAL_ERROR "Offline operator analysis requires Clang; set DAG_LLVM_COMPILER")
endif()
set(DAG_OPERATOR_SOURCES
  "${CMAKE_CURRENT_SOURCE_DIR}/vendor/demo/src/base_kernels/kernels_impl.cpp;${CMAKE_CURRENT_SOURCE_DIR}/vendor/demo/src/base_kernels/kernels_public.cpp"
  CACHE STRING "Operator implementation sources for offline Clang analysis")
set(DAG_OPERATOR_INCLUDES
  "${CMAKE_CURRENT_SOURCE_DIR}/vendor/demo/include;${CMAKE_CURRENT_SOURCE_DIR}/vendor/demo/src/base_kernels"
  CACHE STRING "Include directories for offline operator analysis")
set(DAG_OPERATOR_CLANG_ARGS "" CACHE STRING "Extra Clang flags for offline operator analysis")
set(operator_analysis_args --compiler "${DAG_LLVM_COMPILER}")
foreach(source IN LISTS DAG_OPERATOR_SOURCES)
  list(APPEND operator_analysis_args --operator-source "${source}")
endforeach()
foreach(directory IN LISTS DAG_OPERATOR_INCLUDES)
  list(APPEND operator_analysis_args --include "${directory}")
endforeach()
foreach(flag IN LISTS DAG_OPERATOR_CLANG_ARGS)
  list(APPEND operator_analysis_args "--clang-arg=${flag}")
endforeach()
set(operator_headers)
foreach(directory IN LISTS DAG_OPERATOR_INCLUDES)
  file(GLOB_RECURSE headers CONFIGURE_DEPENDS "${directory}/*.h" "${directory}/*.hpp" "${directory}/*.hh" "${directory}/*.inc")
  list(APPEND operator_headers ${headers})
endforeach()
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/block_cholesky.meta"
  BYPRODUCTS "${CMAKE_CURRENT_BINARY_DIR}/catalog.generated.json"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/compiler/compile.py"
  "${CMAKE_CURRENT_SOURCE_DIR}/vendor/demo/src/baseline/block_cholesky.cpp"
  ${operator_analysis_args}
  --catalog-output "${CMAKE_CURRENT_BINARY_DIR}/catalog.generated.json"
  --encoding "${DAG_METADATA_ENCODING}"
  -o "${CMAKE_CURRENT_BINARY_DIR}/block_cholesky.meta"
  DEPENDS compiler/compile.py compiler/infer_catalog.py ${DAG_OPERATOR_SOURCES} ${operator_headers}
          vendor/demo/src/baseline/block_cholesky.cpp
  VERBATIM)
add_custom_target(metadata ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/block_cholesky.meta")
add_dependencies(dag_run metadata)
